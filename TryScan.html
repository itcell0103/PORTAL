/** * Serves the HTML file as a Web App. */ function doGet() { return HtmlService.createHtmlOutputFromFile('Index') .setTitle('Document Scanner App') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } /** * Saves the generated PDF to the target folder. * @param {string} base64Data The base64 encoded PDF string. * @return {object} Success status, message, and updated files list. */ function savePDFToServer(base64Data) { try { var folderName = "AST_ScannedDocuments"; var folders = DriveApp.getFoldersByName(folderName); var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName); // Format Date & Time: AST-DDMMYY-HHMMSS.pdf var date = new Date(); var dd = String(date.getDate()).padStart(2, '0'); var mm = String(date.getMonth() + 1).padStart(2, '0'); var yy = String(date.getFullYear()).slice(-2); var hh = String(date.getHours()).padStart(2, '0'); var min = String(date.getMinutes()).padStart(2, '0'); var ss = String(date.getSeconds()).padStart(2, '0'); var fileName = 'AST-' + dd + mm + yy + '-' + hh + min + ss + '.pdf'; // Decode and save file var decodedData = Utilities.base64Decode(base64Data.split(',')[1]); var blob = Utilities.newBlob(decodedData, 'application/pdf', fileName); var file = folder.createFile(blob); // Create file metadata object var fileInfo = { id: file.getId(), name: fileName, url: file.getDownloadUrl(), date: date.getTime() }; // Update PropertiesService (Keep last 5) - Fixed to use getScriptProperties var props = PropertiesService.getScriptProperties(); var recentFiles = JSON.parse(props.getProperty('recentPDFs') || '[]'); recentFiles.unshift(fileInfo); if (recentFiles.length > 5) recentFiles = recentFiles.slice(0, 5); props.setProperty('recentPDFs', JSON.stringify(recentFiles)); return { success: true, message: 'Saved as ' + fileName, files: recentFiles }; } catch (e) { return { success: false, message: e.toString() }; } } /** * Retrieves the last 5 processed files. */ function getRecentFiles() { // Fixed to use getScriptProperties var props = PropertiesService.getScriptProperties(); return JSON.parse(props.getProperty('recentPDFs') || '[]'); } /** * Deletes a specific file by ID and updates the recent list. */ function deleteFile(fileId) { try { DriveApp.getFileById(fileId).setTrashed(true); // Fixed to use getScriptProperties var props = PropertiesService.getScriptProperties(); var recentFiles = JSON.parse(props.getProperty('recentPDFs') || '[]'); recentFiles = recentFiles.filter(function(f) { return f.id !== fileId; }); props.setProperty('recentPDFs', JSON.stringify(recentFiles)); return { success: true, files: recentFiles }; } catch (e) { return { success: false, message: e.toString() }; } }
